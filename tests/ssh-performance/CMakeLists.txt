cmake_minimum_required(VERSION 3.24)
project(LogAnalyQtSshPerformance LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_EXPORT_NO_PACKAGE_REGISTRY ON)
get_filename_component(APP_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(SSH_SOURCE_DIR "${APP_SOURCE}/src/core" CACHE PATH "SshTerminal source snapshot to benchmark")
find_package(Qt6 6.11 REQUIRED COMPONENTS Core Gui Network Quick QuickControls2 Test)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_ZLIB_COMPRESSION OFF CACHE BOOL "" FORCE)
if(WIN32)
    set(CRYPTO_BACKEND WinCNG CACHE STRING "" FORCE)
else()
    set(CRYPTO_BACKEND OpenSSL CACHE STRING "" FORCE)
endif()
add_subdirectory("${APP_SOURCE}/third_party/libssh2" libssh2)
add_executable(ssh_performance
    ssh_performance.cpp
    "${SSH_SOURCE_DIR}/SshTerminal.cpp"
    "${SSH_SOURCE_DIR}/SshTerminal.h"
    "${APP_SOURCE}/src/core/SshLogStream.cpp"
    "${APP_SOURCE}/src/core/SecureCredentialStore.cpp")
target_include_directories(ssh_performance PRIVATE "${SSH_SOURCE_DIR}" "${APP_SOURCE}/src" "${APP_SOURCE}/src/core")
target_link_libraries(ssh_performance PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Network Qt6::Test libssh2_static)
if(NOT WIN32)
    find_package(Iconv REQUIRED)
    target_link_libraries(ssh_performance PRIVATE Iconv::Iconv)
endif()
if(APPLE)
    target_link_libraries(ssh_performance PRIVATE "-framework Security" "-framework CoreFoundation")
elseif(WIN32)
    target_link_libraries(ssh_performance PRIVATE Advapi32 Psapi)
endif()

add_executable(ssh_stream_tests "${APP_SOURCE}/tests/ssh_stream_tests.cpp"
    "${APP_SOURCE}/src/core/SshTerminal.cpp" "${APP_SOURCE}/src/core/SshTerminal.h"
    "${APP_SOURCE}/src/core/SshLogStream.cpp" "${APP_SOURCE}/src/core/SecureCredentialStore.cpp")
target_include_directories(ssh_stream_tests PRIVATE "${APP_SOURCE}/src" "${APP_SOURCE}/src/core")
target_link_libraries(ssh_stream_tests PRIVATE Qt6::Gui Qt6::Network Qt6::Test libssh2_static)
if(APPLE)
    target_link_libraries(ssh_stream_tests PRIVATE Iconv::Iconv "-framework Security" "-framework CoreFoundation")
elseif(WIN32)
    target_link_libraries(ssh_stream_tests PRIVATE Advapi32)
else()
    target_link_libraries(ssh_stream_tests PRIVATE Iconv::Iconv)
endif()
enable_testing()
add_test(NAME ssh_stream_tests COMMAND ssh_stream_tests)

foreach(check IN ITEMS ssh_success_session_tests ssh_security_tests ssh_stream_session_tests)
    add_executable(${check} "${APP_SOURCE}/tests/${check}.cpp"
        "${APP_SOURCE}/src/core/SshTerminal.cpp" "${APP_SOURCE}/src/core/SshTerminal.h"
        "${APP_SOURCE}/src/core/SshLogStream.cpp" "${APP_SOURCE}/src/core/SecureCredentialStore.cpp")
    target_include_directories(${check} PRIVATE "${APP_SOURCE}/src" "${APP_SOURCE}/src/core")
    target_link_libraries(${check} PRIVATE Qt6::Gui Qt6::Network libssh2_static)
    if(APPLE)
        target_link_libraries(${check} PRIVATE Iconv::Iconv "-framework Security" "-framework CoreFoundation")
    elseif(WIN32)
        target_link_libraries(${check} PRIVATE Advapi32)
    else()
        target_link_libraries(${check} PRIVATE Iconv::Iconv)
    endif()
endforeach()
